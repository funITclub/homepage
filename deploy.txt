# funITclub デプロイ手順（Azure App Service）
#
# 対象リソース（.azure/config の既定値。--resource-group などは省略できる）
#   App Service : funITclub（PYTHON|3.11 / Linux）
#   リソースグループ : hirahira
#   App Service プラン : hirahiratriangle_asp_1479
#   ロケーション : japanwest
#   PostgreSQL : hirahira-db（公開アクセス無効。VNet 内からのみ）
#
# hirahira-room と同じリソースグループ・同じ DB サーバーを間借りしている。
# ただしデータベースは funITclub 専用（funitclub）を使い、hirahira_room の
# 共有DBには触らないこと。


########################################
# 通常のデプロイ（PR を main にマージするだけ）
########################################

# main には直接 push できない（下の「main の保護と書き込み権」）。ブランチを切って PR を出し、
# テストが通ってマージすると、GitHub Actions（.github/workflows/deploy.yml）が
# zip を作って App Service に送る。完了まで2〜3分。
#   - docs/・README.md・deploy.txt だけの変更ではデプロイしない
#   - 同じものをもう一度送りたいときは GitHub の Actions →「Deploy」→「Run workflow」
#   - 送るのはコミット済みの中身だけ。作業ツリーの未コミットの変更は上がらない
#   - デプロイが重なったら、先のものが終わるまで次は待つ
#
# 結果は GitHub の Actions タブで見る。失敗したら下の「状態の確認」とログを見る。
#
# migrate と createcachetable、seed_fe（FE 対策アプリの分類）は、アプリの起動時に
# startup.sh が実行する。デプロイのたびにコンテナが作り直されるので、手で流す必要はない。
#   - migrate が失敗するとアプリは起動しない。デプロイも失敗扱いになり、
#     GitHub Actions が赤くなる。原因はログ（az webapp log tail）の
#     「startup.sh:」の行のあたりを見る。
#   - GitHub のランナーからは DB（VNet の中）に届かないので、ワークフローでは流さない。

# 状態の確認
az webapp show --name funITclub --query state

# 反映されないときは再起動
az webapp restart --name funITclub

# ログを見る（問題があるとき）
az webapp log tail --name funITclub

# 自動デプロイの設定（初回のみ・構築済み）
########################################

# GitHub Actions から Azure へは OIDC（フェデレーション資格情報）でログインする。
# パスワードや発行プロファイルは使わない。
#
# ※ SCM の基本認証はオフになっている。発行プロファイル方式は動かないが、
#   オンに戻さないこと。OIDC なら基本認証は要らない。
# ※ ポータルの「デプロイ センター」は使わない。自前のワークフローを main に
#   直接コミットしてくるうえ、ランナー上でビルドし docs/ も除外しない。
#
# ---- 現在の構成 ----
#   マネージド ID           : funitclub-github-deploy（hirahira / Japan West）
#                             クライアント ID 4cc3caab-2f2f-439b-9918-09b16392efec
#   フェデレーション資格情報 : github-production
#                             subject repo:funITclub@316769444/homepage@1333670349:environment:production
#   ロール                  : Web サイト共同作成者（App Service funITclub スコープのみ）
#   GitHub の環境           : production（使えるのは main だけ）
#   環境の secrets          : AZURE_CLIENT_ID / AZURE_TENANT_ID / AZURE_SUBSCRIPTION_ID
#                             （WG_READ_APP_ID / WG_READ_APP_PRIVATE_KEY / COUNTDOWN_NOTIFY_TOKEN もここ）
#
# 秘密情報はリポジトリの secrets に置かない。リポジトリに Write があれば、別のブランチで
# ワークフローを書き換えて push するだけで読めてしまう。環境 production は main からしか
# 使えず、main は PR でしか変えられない（下の「main の保護と書き込み権」）。
#   スタートアップ コマンド : bash startup.sh（起動時に migrate → gunicorn）
#
# ---- 作り直すときの手順（Azure ポータル） ----
#
# 1. マネージド ID を作る
#    検索「マネージド ID」→「作成」
#      リソースグループ : hirahira
#      リージョン       : Japan West
#      名前             : funitclub-github-deploy
#
# 2. 環境 production を信頼させる
#    （先に GitHub の funITclub/homepage → Settings → Environments で production を作り、
#      Deployment branches and tags を「Selected branches and tags」にして main だけを足す）
#    funitclub-github-deploy →「設定」→「フェデレーション資格情報」→「資格情報の追加」
#      シナリオ      : GitHub Actions による Azure リソースのデプロイ
#      組織          : funITclub
#      組織 ID       : 316769444
#      リポジトリ    : homepage
#      リポジトリ ID : 1333670349
#      エンティティ  : 環境
#      環境          : production
#      名前          : github-production
#    ※ このリポジトリの OIDC トークンは ID 入りのサブジェクト（use_immutable_subject）を
#      使う。名前を変えても ID は変わらないので、なりすましたリポジトリを弾ける。
#      サブジェクト識別子は手で編集せず、自動生成の
#      repo:funITclub@316769444/homepage@1333670349:environment:production のままにする。
#      1文字でも違うとログインに失敗する。
#    ※ ID の調べ方
#      gh api repos/funITclub/homepage --jq '{owner_id: .owner.id, repo_id: .id}'
#      gh api repos/funITclub/homepage/actions/oidc/customization/sub   # sub_claim_prefix
#    ※ production を付けたジョブ以外、main 以外のブランチからはログインできない（それで正しい）。
#
# 3. デプロイの権限を付ける
#    App Service funITclub →「アクセス制御 (IAM)」→「追加」→「ロールの割り当ての追加」
#      ロール   : Web サイト共同作成者（Website Contributor）
#      割り当て先 : マネージド ID → funitclub-github-deploy
#    ※ リソースグループ hirahira には付けないこと。hirahira-room・hirahira-db・
#      Key Vault まで触れるようになる。アプリ1つに絞る。
#
# 4. funitclub-github-deploy の「概要」でクライアント ID を控える
#
# 5. GitHub の funITclub/homepage → Settings → Environments → production の
#    Environment secrets に3つ登録する（リポジトリの管理者権限が要る）。
#    gh なら gh secret set <名前> --repo funITclub/homepage --env production
#      AZURE_CLIENT_ID       : 4. のクライアント ID
#      AZURE_TENANT_ID       : az account show --query tenantId -o tsv
#      AZURE_SUBSCRIPTION_ID : az account show --query id -o tsv
#
# 6. スタートアップ コマンドを設定する（起動時に migrate を流すため）
#    App Service funITclub →「設定」→「構成」→「スタックの設定」
#      スタートアップ コマンド : bash startup.sh
#    または
#      az webapp config set --name funITclub --startup-file "bash startup.sh"
#    ※ 設定すると再起動する。startup.sh がデプロイ済みでないと起動に失敗するので、
#      先に startup.sh を含むコミットをデプロイしてから設定すること。
#    ※ 戻すときは空にする（Oryx の自動生成の起動に戻る。migrate は流れなくなる）。
#      az webapp config set --name funITclub --startup-file ""
#
# 確認（資格情報とロールが入っているか）
az identity federated-credential list --identity-name funitclub-github-deploy \
  --resource-group hirahira --query "[].subject"
#   ※ --all は .azure/config の既定の group とぶつかってエラーになるので、scope で指定する。
az role assignment list --assignee 4cc3caab-2f2f-439b-9918-09b16392efec \
  --scope "$(az webapp show --name funITclub --query id -o tsv)" \
  --query "[].{role:roleDefinitionName,scope:scope}"


########################################
# main の保護と書き込み権
########################################

# main に入ったものはそのまま本番に出るので、main は PR でしか変えられないようにしてある。
#
# ---- 現在の構成 ----
#   書き込み権   : チーム homepage に Write。人ごとには付けない（卒業で入れ替わるため）。
#                  org のオーナーは admin のまま
#   ルールセット : main（funITclub/homepage → Settings → Rules → Rulesets）
#                  - PR 必須・承認 1 件
#                  - テスト（Test ワークフローの test）が通らないとマージできない
#                  - 直接 push・force push・ブランチの削除は禁止
#                  - 迂回できるのは admin だけ、それも PR の上でだけ（直接 push はできない）
#   マージしたブランチは自動で消す
#
# ※ チームの名前を wg- で始めないこと。WG への参加の仕組みは wg- のチームに誰でも
#   入れるので（wgjoin/links.py）、運営のチームが紛れると誰でも homepage に書けてしまう。
# ※ 承認 1 件は、書き込める人が自分ひとりのときは満たせない。そのときは admin が
#   gh pr merge <番号> --admin で迂回してマージする（テストが通ってから）。
#
# 人を足す・外す
#   gh api -X PUT orgs/funITclub/teams/homepage/memberships/<ユーザー名> -f role=member
#   gh api -X DELETE orgs/funITclub/teams/homepage/memberships/<ユーザー名>
#   （org の管理者権限が要る。gh は gh auth refresh -h github.com -s admin:org で権限を足す）
#
# 確認
gh api repos/funITclub/homepage/rulesets --jq '.[]|[.name,.enforcement]'
gh api repos/funITclub/homepage/teams --jq '.[]|[.slug,.permission]'


########################################
# 手動のデプロイ（GitHub Actions が使えないとき）
########################################

# zip デプロイを使う。アプリのコンテンツだけを差し替えるので、
# App Service プランには触らない。SCM_DO_BUILD_DURING_DEPLOYMENT=True なので
# デプロイ時に pip install と collectstatic が走る。
#
# ※ az webapp up は使えない。あれは毎回 App Service プランを
#    create-or-update しようとするが、B1（Basic）のプランに Regional VNet 統合が
#    付いていると、同じ B1 への更新であっても Azure に拒否される
#    （Conflict / ExtendedCode 04519）。
#    エラーは「VNet 統合を切ってから再実行せよ」と案内してくるが、切らないこと。
#    hirahira-db への唯一の経路がこの VNet 統合で、外すと DB に繋がらなくなる。

# 1. 送るものを zip に固める。リポジトリのルートで実行する。
#    .gitignore を除外リストとして使うので、venv・db.sqlite3・staticfiles・
#    __pycache__ は自動的に除外される。除外条件を二重に管理しなくて済む。
#    コミットの有無は関係ない。作業ツリーの中身がそのまま上がる
#    （--others で未追跡ファイルも含め、--exclude-standard で .gitignore を効かせる）。
#    ※ zip は既存ファイルに「追記」する。前回の残骸があると古いファイルが
#      混ざるので、必ず先に消してから作ること。
rm -f /tmp/funitclub-deploy.zip
git ls-files --cached --others --exclude-standard -z \
  | xargs -0 zip -q -X /tmp/funitclub-deploy.zip

#    docs/ は運用資料、.github/ は自動デプロイの設定で、アプリの動作には
#    関係ないので落とす。.gitignore に入れるとリポジトリからも消えてしまうため、ここで除外する。
zip -q -d /tmp/funitclub-deploy.zip 'docs/*' '.github/*'

# 2. 中身を確認する（venv や db.sqlite3 が混ざっていないこと）
unzip -l /tmp/funitclub-deploy.zip

# 3. デプロイする。完了まで2分ほどかかる。
az webapp deploy --name funITclub --resource-group hirahira \
  --src-path /tmp/funitclub-deploy.zip --type zip

# 4. 使い終わった zip を消す
rm /tmp/funitclub-deploy.zip

# 初回のみ（環境の作成）
########################################

# 1. funITclub 専用データベースを作る。
#    ※ hirahira_room の共有DBは使わない。
az postgres flexible-server db create --resource-group hirahira --server-name hirahira-db --database-name funitclub --charset UTF8 --collation en_US.utf8

# 2. SECRET_KEY を生成する。
python -c "import secrets; import string; chars = string.ascii_letters + string.digits + '-_'; print(''.join(secrets.choice(chars) for _ in range(50)))"

# 3. アプリケーション設定（環境変数）を入れる。
#    DB_USER / DB_PASSWORD / DB_HOST は hirahira-room と同じ値。
#    DB_NAME だけ funITclub 専用DBを指すこと。
az webapp config appsettings set --name funITclub --settings \
  SECRET_KEY="<生成した値>" \
  DB_NAME="funitclub" \
  DB_USER="<ユーザー名>" \
  DB_PASSWORD="<パスワード>" \
  DB_HOST="<hirahira-db のホスト名>" \
  SCM_DO_BUILD_DURING_DEPLOYMENT="True"

# 3-2. 参加フォーム（/join/apply/）の送信設定は「メール送信」の節を参照。
#    ACS を作ってから EMAIL_HOST_USER / EMAIL_HOST_PASSWORD を設定する。

# 4. VNet 統合（hirahira-db は VNet 内からしか繋がらない）。
#    hirahira-room と同じ hirahara_vnet の中に、funITclub 専用サブネットを切って繋ぐ。
#    ※ 別VNetにすると hirahira-db に届かなくなる（ピアリングとプライベートDNSが別途必要）。
az network vnet subnet create --resource-group hirahira --vnet-name hirahara_vnet --name funitclub_vnet_apps --address-prefixes 10.1.2.0/24 --delegations Microsoft.Web/serverfarms
az webapp vnet-integration add --name funITclub --resource-group hirahira --vnet hirahara_vnet --subnet funitclub_vnet_apps

#    サブネットの割り当て（現状）
#      hirahira_vnet_db     10.1.0.0/24  PostgreSQL
#      hirahira_vnet_apps   10.1.1.0/24  hirahira-room
#      funitclub_vnet_apps  10.1.2.0/24  funITclub

# 5. 編集画面のアカウントを作る。
#    App Service の SSH から実行する（手元からは hirahira-db に繋がらない）。
#    ※ migrate と createcachetable（参加フォームの連続送信の検知用テーブル）は、
#      アプリの起動時に startup.sh が流す。スタートアップ コマンドの設定は
#      「自動デプロイの設定」の 6. を参照。
python manage.py createsuperuser

# 6. 管理者（通知の宛先・公開の問い合わせ先）を確認する。
#    migrate で公開の問い合わせ先が登録される。増減は /admin/ →「管理者」。
#    ※ 設定ファイルやテンプレートにアドレスを書かないこと。参照先はこのテーブルだけ。


########################################
# メール送信（参加フォーム /join/apply/）
########################################

# Azure Communication Services（ACS）の Email を使い、独自ドメイン funitclub.org の
# 差出人（no-reply@funitclub.org）で送る。経路・認証の仕組みと、大学のアカウントから
# 送らない理由は docs/design.md の「1. メール送信の仕組み」。ここは構成と手順だけ。
#
# funitclub.org の DNS は Azure DNS（リソースグループ hirahira のゾーン funitclub.org）に
# あるので、検証用レコードも az で足せる。

# ---- 現在の構成（構築済み。作り直すときだけ下の手順を使う） ----
#   Email Communication Service : funitclub-email（data-location: Japan）
#   送信ドメイン                : funitclub.org（CustomerManaged / Domain・SPF・DKIM・DKIM2 検証済み）
#   差出人                      : no-reply@funitclub.org
#   Communication Service       : funitclub-comm（funitclub.org をリンク済み）
#   Entra アプリ                : funitclub-smtp
#                                 appId bdd9eab1-9a2d-4f46-be1d-47e7131bd2ff
#                                 カスタムロール "funITclub Email Sender"（funitclub-comm スコープ）
#                                 ※ 組み込みの "Communication and Email Service Owner" は
#                                   広すぎるので外してある。メール送信に必要な3つの操作だけ。
#   SMTP ユーザー名             : funitclub-smtp（username: no-reply@funitclub.org）
#   Key Vault                   : funitclub-kv（RBAC 認可）
#                                 シークレット funitclub-smtp-password
#   App Service の設定          : EMAIL_HOST_USER は平文、EMAIL_HOST_PASSWORD は Key Vault 参照
#                                 （システム割り当てマネージド ID に "Key Vault Secrets User"）
#
#   追加した DNS レコード（funitclub.org ゾーン）
#     TXT   @                                        ms-domain-verification=37abf404-...
#     TXT   @                                        v=spf1 include:spf.protection.outlook.com -all
#     TXT   _dmarc                                   v=DMARC1; p=none; rua=mailto:contact@funitclub.org
#     CNAME selector1-azurecomm-prod-net._domainkey  selector1-azurecomm-prod-net._domainkey.azurecomm.net
#     CNAME selector2-azurecomm-prod-net._domainkey  selector2-azurecomm-prod-net._domainkey.azurecomm.net
#
#   クライアント シークレットの有効期限は2年（2028-08-19 まで）。
#   30日前から管理者に通知が届く（EMAIL_SECRET_EXPIRES_ON を見ている）。
#   切れる前に発行し直し、Key Vault と期限日の設定を更新する。
#   App Service の設定はバージョン無しの参照なので、Key Vault を更新すれば追従する
#   （反映は最大24時間。すぐ反映したいなら az webapp restart）。
#     az ad app credential reset --id bdd9eab1-9a2d-4f46-be1d-47e7131bd2ff --years 2
#     az keyvault secret set --vault-name funitclub-kv --name funitclub-smtp-password --value "<新しい値>"
#     az webapp config appsettings set --name funITclub --settings EMAIL_SECRET_EXPIRES_ON="<新しい期限>"
#     ※ 期限日の更新を忘れると通知が嘘になる。
#
#   Key Vault 参照が解決できているかの確認
#     az rest --method get --url "https://management.azure.com/subscriptions/$(az account show --query id -o tsv)\
# /resourceGroups/hirahira/providers/Microsoft.Web/sites/funITclub/config/configreferences/appsettings?api-version=2022-03-01"
#
#   DMARC は現在 p=none（監視のみ）。しばらくレポートを見て問題なければ
#   p=quarantine → p=reject と段階的に強めると、funitclub.org を騙るメールを弾ける。
#   ※ ACS の DMARC 検証（initiate-verification --verification-type DMARC）はサービス側が
#     受け付けない。DNS にレコードがあれば機能するので、検証状態 NotStarted のままでよい。
#
#   連続送信の検知・遮断のログを見る（WARNING で出る）
#     az webapp log tail --name funITclub | grep -e 連続送信 -e 遮断
#     手動の解除は /admin/ の「遮断IP」→「今すぐ解除する」（遮断の仕組みは docs/design.md の 2-4）。
#
#   自分のIPを誤って遮断して admin に入れなくなったときの復旧（DB を触らずに戻せる）
#     az webapp config appsettings set --name funITclub --settings IP_BLOCK_EXEMPT="<自分のIP>"
#     az webapp config appsettings set --name funITclub --settings IP_BLOCK_ENABLED="False"
#     ※ 復旧して遮断を消したら、この設定は消しておくこと。
#     az webapp config appsettings delete --name funITclub --setting-names IP_BLOCK_EXEMPT
#
#   メールを送らずに認証だけ確かめる（手元から）
#     python -c "import os,smtplib; s=smtplib.SMTP('smtp.azurecomm.net',587); s.starttls(); \
#       s.login('no-reply@funitclub.org', os.environ['EMAIL_HOST_PASSWORD']); print('ok'); s.quit()"

# ---- 初回のみ（送信環境の作成） ----

# 1. Email Communication Service（ドメインを持つ側）を作る。
#    data-location は Japan。グローバルリソースなので location は global。
az communication email create --name funitclub-email --resource-group hirahira \
  --location global --data-location Japan

# 2. 独自ドメイン funitclub.org を登録する（送信者ドメイン）。
az communication email domain create --domain-name funitclub.org \
  --email-service-name funitclub-email --resource-group hirahira \
  --location global --domain-management CustomerManaged

# 3. 検証用の DNS レコードを確認する（Domain / SPF / DKIM / DKIM2 の4種類が出る）。
az communication email domain show --domain-name funitclub.org \
  --email-service-name funitclub-email --resource-group hirahira \
  --query verificationRecords

# 4. 出てきた値を Azure DNS のゾーンに足す。TXT は2件、CNAME は2件。
#    ※ SPF は既存の TXT があるなら1本にまとめること（v=spf1 は1ドメイン1レコード）。
az network dns record-set txt add-record --resource-group hirahira --zone-name funitclub.org \
  --record-set-name "@" --value "<Domain の TXT 値>"
az network dns record-set txt add-record --resource-group hirahira --zone-name funitclub.org \
  --record-set-name "@" --value "v=spf1 include:spf.protection.outlook.com -all"
az network dns record-set cname set-record --resource-group hirahira --zone-name funitclub.org \
  --record-set-name "selector1-azurecomm-prod-net._domainkey" --cname "<DKIM の値>"
az network dns record-set cname set-record --resource-group hirahira --zone-name funitclub.org \
  --record-set-name "selector2-azurecomm-prod-net._domainkey" --cname "<DKIM2 の値>"

#      gh secret set COUNTDOWN_NOTIFY_TOKEN --repo funITclub/homepage --env production
#

# WG のアプリを載せる（/<アプリ名>/）
########################################

# WG のアプリは WG のリポジトリ（非公開）で作り、homepage には入れない。デプロイのときだけ
# wg/<名前>/ に取り込んで App Service に載せる。仕組み（なぜ取り込むのか・組み込み方・
# requirements の合流・通行証）は README の「サブアプリ（works/・wg/）」。ここは手順だけ。
#
#   funITclub/wg-<名前>（非公開）         ソースの置き場。ここで開発する
#        ↓ デプロイのときだけ取り込む（wg-apps.yml）
#   App Service funITclub の wg/<名前>/   公開はここで動く。URL は /<名前>/
#
# 公表できる形になったら as を works に変えるだけでよい（URL は変わらない）。
#
# ---- 使う GitHub App（初回のみ） ----
# 1つの App で足りる。権限は次の2つだけにする。
#   WG のリポジトリ  Contents: Read    homepage がデプロイ時にソースを取り込む
#   homepage         Actions:  Write   WG 側から homepage のデプロイを起動する
#
# ※ homepage に Contents: Write は付けない。WG のリポジトリは学生のメンバーが触れるので、
#   そこに置いた資格情報で homepage のコードを書き換えられてはいけない。
#   Actions: Write では「このワークフローを起動する」ことしかできない。
#
# 1. org の Settings → Developer settings → GitHub Apps → New GitHub App
#      名前: funitclub-wg-deploy
#      Homepage URL: https://funitclub.org
#      Webhook: Active のチェックを外す
#      Repository permissions: Contents = Read-only、Actions = Read and write
#      Where can this GitHub App be installed?: Only on this account
# 2. 作成後の画面（General）で App ID を控え、左のメニューの Credentials →
#    Private keys → Generate a private key（.pem が落ちる）。秘密鍵とクライアント
#    シークレットは General ではなく Credentials にある
# 3. Install App → funITclub → Only select repositories で homepage と各 WG を選ぶ
#    （WG を増やしたら、ここに足す）
# 4. シークレットを入れる（.pem はそのまま貼る。改行も含める）
#      gh secret set WG_READ_APP_ID --repo funITclub/homepage --env production --body "<App ID>"
#      gh secret set WG_READ_APP_PRIVATE_KEY --repo funITclub/homepage --env production < <鍵のファイル>
#      gh secret set HOMEPAGE_DEPLOY_APP_ID --repo funITclub/wg-<名前> --body "<App ID>"
#      gh secret set HOMEPAGE_DEPLOY_PRIVATE_KEY --repo funITclub/wg-<名前> < <鍵のファイル>
# 5. 入れ終わったら .pem をローカルから消す（鍵は作り直せる）
#
# 鍵を作り直すとき：2. の Credentials で新しい鍵を作り、4. の homepage と WG すべての
# PRIVATE_KEY を入れ替える。デプロイが通るのを確かめてから、古い鍵を Credentials で消す。
#
# ---- WG を1つ載せる ----
# 1. WG のリポジトリに .github/workflows/deploy.yml を置く（wg-template に入っている。
#    main への push で homepage のデプロイを起動する）
# 2. homepage の wg-apps.yml に足す
#      apps:
#        - repo: funITclub/wg-protein
#          take:
#            - protein                                          # → /protein/（フォルダー名がそのまま URL）
#            - {app: protein_activity, url: protein/activity}   # → /protein/activity/ 活動ページ
#          as: wg                # 公表したら works に変える
#    1つのリポジトリに複数のアプリがあるときは take に並べる。URL を入れ子にするときは
#    {app, url} と書く（書き方は wg-apps.yml の先頭）。活動ページだけを載せてもよい。
#    並べなかったものは本番に載らないので、載せるアプリから載せないアプリへリンクしないこと。
# 3. main に push する（homepage 側）。デプロイが走り、/protein/ が生える
# 4. 編集画面（/edit/）の WG紹介で「Webアプリの URL」に https://funitclub.org/protein/ を入れる
#    （これを入れるまで、一覧から辿る導線が無いので誰も開けない）。活動ページ（/protein/activity/）
#    は「追加リンクの名前」を「活動ページ」にして、「追加リンクの URL」に入れる

# ---- 気をつけること ----
#   - アプリ側に apps.py を置かない（または name を書かない）。WG のリポジトリでは
#     <名前>、本番では wg.<名前> と読み込まれる名前が変わるため。無ければ Django が
#     自動で作り、呼び名（label）はどちらも <名前> になるのでマイグレーションが一致する。
#   - WG の requirements.txt は本番の分に合流し、同じパッケージは本番の版が勝つ。
#     食い違いはデプロイのログに warning で出る。
#   - 初回のデプロイで、WG アプリのテーブルが本番 DB にできる（起動時の migrate）。
#     あとから消すのは手作業になるので、モデルの形を決めてから載せる。
#   - 出す前のテストと、出した後の応答の確認・自動の巻き戻しは README の「開発と本番の切り替え」。
#     止まったら Actions のログの「Check the site with the WG apps」などを見る。
#   - WG のアプリが出した ERROR のログも、運営に通知メールで届く（ログはルートのロガーで受けている）。
#     ログにアドレスや個人の情報を書かないよう、WG に伝えておく。
#   - WG が終わったら wg-apps.yml から消す。次のデプロイで本番から外れる（DB は残る）。
#
# ---- 確認 ----
#   取り込まれたか（デプロイのログの「Take in the WG apps」に 〜@<コミット> が出る）
#     gh run view --repo funITclub/homepage --log | grep "→ wg/"
#   直打ちで開けないこと（一覧へ送り返される）

# WG の活動ページのチケット（GitHub の読み取り）
########################################

# 活動ページ（/<WG>/activity/、funITclub/claude-plugins の wg-activity スキル）は、チケットを
# WG のリポジトリの Issue と org の Project から読んで出す。書くのは会員が GitHub で直接行い、
# 活動ページには書き込む画面が無い。題名と本文は一般公開する（担当者・コメント・画像は出さない）。
#
# 読むのは「WG への参加」と同じ GitHub App funitclub-wg-join。鍵も同じ（Key Vault の
# funitclub-github-app-private-key）で、App Service の設定は増えない。トークンは読むたびに、
# その WG のリポジトリの Issues（と org の Projects）の読み取りだけに絞って取る
# （活動ページのトークンに Members の書き込みは付かない）。読んだものは DB キャッシュ
# （funitclub_cache）に 5 分置き、読めなかったら最後に読めた分（1 日とっておく）を出す。
#
# ※ funitclub-wg-deploy（デプロイの取り込み用）は使わない。その鍵は WG のリポジトリの
#   シークレットにも置いてあり、学生が触れる場所のため。
#
# ---- App に権限を足す（初回のみ） ----
# 1. org funITclub の Settings → Developer settings → GitHub Apps → funitclub-wg-join →
#    Permissions & events
#      Repository permissions:   Issues = Read-only（Metadata = Read-only は自動で付く）
#      Organization permissions: Projects = Read-only
#    → Save changes
# 2. org の Settings → GitHub Apps → funitclub-wg-join → Configure
#      - 権限が増えた知らせの「Review request」→「Accept new permissions」
#      - Repository access を All repositories にする（WG が増えても足さなくてよい）
# 3. 下の「確認」で読めることを確かめる
#   権限が入る前に活動ページを出しても、「いま GitHub からチケットを読めません」と出るだけで、
#   紹介と構想資料は出る（ログに WARNING。通知メールは来ない）。
#
# ---- Project のテンプレート ----
# ステータス・優先度・期日などを使う WG は、このテンプレートを wg-activity スキルで複製する。
# 使うかは WG が決める（運営は WG ごとの作業をしない）。題名と項目の名前はスキルと活動ページが
# 名前で探すので変えない。
#
#   現在の構成（2026-10-23 に作成。スキルの手順で複製・読み取りまで確かめた）
#     https://github.com/orgs/funITclub/projects/1 「活動ページのテンプレート」（テンプレートの印つき・非公開）
#     項目       Status（新規・進行中・解決・フィードバック・終了・却下）・優先度（低め・通常・高め・急いで）・
#                開始日・期日・進捗率・プロジェクト（選択肢「全体」。複製した WG が直す）
#     Workflows  Item added to project → 新規 / Item closed → 終了 / Item reopened → 進行中 /
#                Auto-close issue（終了 にしたら閉じる）/ Pull request merged（オン）/
#                Pull request linked to issue → オフ
#     複製すると、項目・選択肢・Workflows が引き継がれる（複製はテンプレートの印なしの普通の Project）
#     活動ページは、解決・終了・却下と Closed の Issue を完了とみなし、並べ替えは Status の選択肢の順
#
# 作り直すとき（手元で。gh は org のオーナーのアカウントで）:
gh auth refresh -h github.com -s project
N=$(gh project create --owner funITclub --title "活動ページのテンプレート" --format json --jq .number)
gh project field-create "$N" --owner funITclub --name 優先度 --data-type SINGLE_SELECT --single-select-options "低め,通常,高め,急いで"
gh project field-create "$N" --owner funITclub --name 開始日 --data-type DATE
gh project field-create "$N" --owner funITclub --name 期日 --data-type DATE
gh project field-create "$N" --owner funITclub --name 進捗率 --data-type NUMBER
gh project field-create "$N" --owner funITclub --name プロジェクト --data-type SINGLE_SELECT --single-select-options "全体"
gh project mark-template "$N" --owner funITclub
# Status の選択肢を入れ替える（元の Todo・In Progress・Done を消して並べ直す）
F=$(gh project field-list "$N" --owner funITclub --format json --jq '.fields[] | select(.name=="Status") | .id')
gh api graphql -f fid="$F" -f query='mutation($fid: ID!) {
  updateProjectV2Field(input: {fieldId: $fid, singleSelectOptions: [
    {name: "新規", color: RED, description: "まだ手を付けていない"},
    {name: "進行中", color: BLUE, description: "取りかかっている"},
    {name: "解決", color: GREEN, description: "直した・作った。確かめ待ち"},
    {name: "フィードバック", color: ORANGE, description: "返事・確認を待っている"},
    {name: "終了", color: GRAY, description: "終わった"},
    {name: "却下", color: GRAY, description: "やらないことにした"}]}) { clientMutationId } }'
# Workflows は gh で設定できないので、ブラウザで https://github.com/orgs/funITclub/projects/<N> の
# 右上の … → Workflows を開き、上の「現在の構成」のとおりにする。選択肢を入れ替えたので、
# 元の Done・In Progress を指していた Auto-close issue と Pull request linked to issue は赤い印に
# なっている。値を選び直すか、オフにする。
#

#   読めなかったときのログ
